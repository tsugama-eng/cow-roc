#For mouse tRNA data
cd DT_PCA
perl -e '
%codon;
%rpm;
$linenum=0;
while($line = <>){
  $linenum++;
  if($linenum == 1) {print $line; next;}
  $line=~s/[\r\n]+//;
  @F = split("\t", $line);
  if(!$rpm{$F[1]}){
    $codon{$F[1]} = $F[2];
    $rpm{$F[1]} = join("\t", @F[3..$#F]);
  } else {
    @rpmdata = split("\t", $rpm{$F[1]}); 
    for $i (0..$#rpmdata) {
      $rpmdata[$i] = $rpmdata[$i] + $F[$i+3];
    }
    $rpm{$F[1]} = join("\t", @rpmdata);
  }
}
for $i (sort keys %rpm) {
  $type = $i;
  $type =~ s/-\d+$//;
  print "$i\t$type\t$codon{$i}\t$rpm{$i}\n";
}
' Mus_musculus_tRNA_RPM.txt > Mus_musculus_tRNA_RPM_mod.txt
