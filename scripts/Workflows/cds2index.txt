for i in Triticum_aestivum Zea_mays Sorghum_bicolor Hordeum_vulgare Cenchrus_americanus Ananas_comosus Musa_acuminata Asparagus_officinalis Allium_cepa Populus_trichocarpa Malus_domestica Vitis_vinifera Citrus_clementina Brassica_rapa Solanum_lycopersicum Glycine_max Chenopodium_quinoa Physcomitrium_patens Chlamydomonas_reinhardtii Bacillus_subtilis Mycobacterium_tuberculosis Salmonella_enterica; do
  perl -nle '
  my ($gene, $mrna);
  if (/^>(\S+).*\bgene:(\S+)/) {
      # 1. "gene:" キーワードが含まれるEnsembl形式など
      ($gene, $mrna) = ($2, $1);
  } elsif (/^>(\S+)/) {
      # 2. キーワードがない場合、先頭のIDを両方に割り当てる（NCBIやUniProtなど）
      ($gene, $mrna) = ($1, $1);
  }
  print "$gene\t$mrna" if $gene;
' ${i}*cds.all.fa > ${i}_gene_mRNA_ID.txt
done

#To make the gene names compatible with further analyses
cd CDS/stAI
#rename 's/stAI\.txt/stAI0.txt/' *.stAI.txt
for i in Arabidopsis*.stAI0.txt Oryza*.stAI0.txt
do
  filename=`echo $i|perl -pe 's/AI0\.txt/AI.txt/'`
  perl -e '
  %stai;
  print "gene\tstAI\n";
  while(<>){
    ($gene, $val) = ($1, $2) if /([\w-]+).*\t(.*?)$/;
    $gene =~ s/^AT/gene:AT/; #For Arabidopsis
    $gene =~ s/^Os(\d+)t(\d+).*/gene:Os$1g$2/; #For rice
    #$gene =~ s/^ENST/gene:ENSG/; #For human
    #$gene =~ s/^ENSMUST/gene:ENSMUSG/; #For mouse
    $stai{$gene} = $val;
  }
  for $i (sort keys %stai){print "$i\t$stai{$i}\n";}
  ' $i > $filename
done

for i in Homo*.stAI0.txt Mus*.stAI0.txt
do
  filename=`echo $i|perl -pe 's/AI0\.txt/AI.txt/'`
  perl -se '
  $idfile = "../Homo_sapiens_gene_mRNA_pep_ID.txt";
  if ($filename =~ /^Mus/){$idfile = "../Mus_musculus_gene_mRNA_pep_ID.txt";}
  %mr2g;
  open(IF, "< $idfile") or die;
  while($line=<IF>){
    ($gid,$mrid,$pid)=split("\t",$line);
    $mr2g{$mrid}=$gid;
  }
  close(IF);
  %stai;
  print "gene\tstAI\n";
  while(<>){
    ($mrna, $val) = ($1, $2) if /([\w\.]+).*\t(.*?)$/;
    $stai{$mr2g{$mrna}} = $val;
  }
  for $i (sort keys %stai){print "gene:$i\t$stai{$i}\n";}
  ' -- -filename=$filename $i > $filename
done

perl -e '
   %tpmdata;
   $linenum=0;
   @samples;
   while($line=<>){
     $linenum++;
     if($linenum==1){@F=split("\t",$line);@samples=@F;next;}
     $line=~s/[\r\n]+//g;
     ($gene,$tpms)=split("\t",$line,2);
     $gene=~s/^gene://;
     $tpmdata{$gene}=$tpms;
   }
   for $i (1..$#samples){
     %tpm = ();
     $outfile= "> Arab_TPM-based_CAI/Arabidopsis_thaliana_tpm_5p.".$i.".txt";
     $valid=0;
     $pcount=0;
     for $j (keys %tpmdata){
       @data = split("\t",$tpmdata{$j});
       $data[$i-1]=~s/[\r\n]+//g;
       if($data[$i-1]>0){$tpm{$j}=$data[$i-1];$valid++;}
     }
     open(OF, $outfile);
     print OF "gene\t${samples[$i]}\n";
     for $j (sort { $tpm{$b} <=> $tpm{$a} } keys %tpm){
        if (int($valid * 0.05) > $pcount){
          print OF "$j\t$tpm{$j}\n";
          $pcount++;
        } else {
          last;
        }
     }
     close(OF);
   }
' TPM/Arabidopsis_thaliana_all_gene_tpm.txt
