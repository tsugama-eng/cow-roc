for i in `awk -F"\t" 'BEGIN{OFS="_";} NR>1{if($1 == "Rice" && $1 != "\#" && ($3 == "mRNA" || $3 == "RNA")){print $2}}' RNA-Seq_data_used.txt`
do
  fasterq-dump --split-files -p $i
  if [ -f ${i}_2.fastq ]; then
    fastp -i ${i}_1.fastq -o ${i}_R1.fastq -I ${i}_2.fastq -O ${i}_R2.fastq
    hisat2 -x ../hisat2db/IRGSP-1.0_genome -1 ${i}_R1.fastq -2 ${i}_R2.fastq -S $i.sam -p 3
    samtools sort -@ 3 -O BAM $i.sam > $i.bam
    samtools index $i.bam
    rm $i.sam ${i}*R*.fastq
    do
      featureCounts -p -T 3 -t gene -g ID -a Oryza_sativa.IRGSP-1.0.59.mod.gff3 -o Os_${i}_gene.count.txt $i.bam
      ../DT_modules/count2tpm.pl -g 1 -l 6 -c 7 Os_${i}_gene.count.txt > Os_${i}_gene.tpm.txt
    done
  else
    fastp -i ${i}.fastq -o ${i}_R.fastq
    hisat2 -x ../hisat2db/IRGSP-1.0_genome -U ${i}_R.fastq -S $i.sam -p 3
    samtools sort -@ 3 -O BAM $i.sam > $i.bam
    samtools index $i.bam
    rm $i.sam ${i}*R*.fastq
    do
      featureCounts -T 3 -t gene -g ID -a Oryza_sativa.IRGSP-1.0.59.mod.gff3 -o Os_${i}_gene.count.txt $i.bam
      ../DT_modules/count2tpm.pl -g 1 -l 6 -c 7 Os_${i}_gene.count.txt > Os_${i}_gene.tpm.txt
    done
  fi
  done
done
