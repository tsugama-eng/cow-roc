EXTERNAL_BAM_DIR="/media/dt/LaCie/codon_project_Ribo-seq/BAM"
RESULT_DIR="/media/dt/LaCie/codon_project_Ribo-seq/RiboCode_Results"

for i in `awk -F"\t" 'BEGIN{OFS=","} {if($10 == 3 && $1 !~ /^y/){gsub(/.*\(|\)/, "", $2); gsub(" ", "_", $2); gsub(" ", "_", $6); print $2,$3,$6}}' further_Ribo-Seq_data.txt`
do
  species=`echo $i | cut -d',' -f1`
  sra=`echo $i | cut -d',' -f2`
  data_type=`echo $i|perl -pe 's/.*(RNA|Ribo).*/$1/'` # RNA or Ribo
  
  if [ ! -f ${species}*dna*.toplevel.fa ]
  then
    sspecies=`echo $species|perl -pe 'tr/A-Z/a-z/'`
    baseurl1="https://ftp.ensemblgenomes.ebi.ac.uk/pub/plants/release-62"
    baseurl2="https://ftp.ensemblgenomes.ebi.ac.uk/pub/metazoa/release-62"
    genome1=`curl ${baseurl1}/fasta/${sspecies}/dna/|perl -nlse 'if(/href="($species.*?dna_sm.toplevel.fa.gz)"/){print "$1\n"};' -- -species=$species`
    cds1=`curl ${baseurl1}/fasta/${sspecies}/cds/|perl -nlse 'if(/href="($species.*?.cds.all.fa.gz)"/){print "$1\n"};' -- -species=$species`
    gff1=`curl ${baseurl1}/gff3/${sspecies}/|perl -nlse 'if(/href="($species.*?.gff3.gz)"/){$cfile = $1; if($cfile !~ /chr/) {print "$cfile\n";}}' -- -species=$species`
    genome2=`curl ${baseurl2}/fasta/${sspecies}/dna/|perl -nlse 'if(/href="($species.*?dna_sm.toplevel.fa.gz)"/){print "$1\n"};' -- -species=$species`
    cds2=`curl ${baseurl2}/fasta/${sspecies}/cds/|perl -nlse 'if(/href="($species.*?.cds.all.fa.gz)"/){print "$1\n"};' -- -species=$species`
    gff2=`curl ${baseurl2}/gff3/${sspecies}/|perl -nlse 'if(/href="($species.*?.gff3.gz)"/){$cfile = $1; if($cfile !~ /chr/) {print "$cfile\n";}}' -- -species=$species`
    wget $baseurl1/fasta/${sspecies}/dna/${genome1}
    wget $baseurl1/fasta/${sspecies}/cds/${cds1}
    wget $baseurl1/gff3/${sspecies}/${gff1}
    wget $baseurl2/fasta/${sspecies}/dna/${genome2}
    wget $baseurl2/fasta/${sspecies}/cds/${cds2}
    wget $baseurl2/gff3/${sspecies}/${gff2}
  fi
  gunzip ${species}*dna*.toplevel.fa.gz ${species}*.gff3.gz

  if [ ! -d ${species}_STAR_index ]; then
    mkdir -p ${species}_STAR_index
    if [ "$species" == "Homo_sapiens" ] || [ "$species" == "Mus_musculus" ]; then
        SA_BASES=14
    elif [ "$species" == "Oryza_sativa" ]; then
        SA_BASES=13
    elif [ "$species" == "Arabidopsis_thaliana" ] || [ "$species" == "Drosophila_melanogaster" ]; then
        SA_BASES=12
    elif [ "$species" == "Saccharomyces_cerevisiae" ]; then
        SA_BASES=10 # 酵母用
    elif [ "$species" == "Escherichia_coli" ]; then
        SA_BASES=10 # 大腸菌用
    else
        # 未知の小型ゲノム用にさらに安全な値を設定
        SA_BASES=10
    fi
    
    cp /home/dt/Downloads/${species}*dna*.toplevel.fa ./
    
    STAR --runMode genomeGenerate --runThreadN 60 \
         --genomeDir ${species}_STAR_index \
         --genomeFastaFiles ${species}*dna*.toplevel.fa \
         --sjdbGTFfile ${species}*.gtf \
         --sjdbOverhang 100 \
         --genomeSAindexNbases $SA_BASES
  fi

  expt=`efetch -db sra -id $sra -format docsum|xtract -pattern DocumentSummarySet -element Experiment@acc`
  sexpt=`echo $expt|perl -pe 's/^(\w{6}).*/$1/'`
  ssexpt=`echo $expt|perl -pe 's/^(\w{3}).*/$1/'`
  submitter=`efetch -db sra -id $sra -format docsum|xtract -pattern DocumentSummarySet -element Submitter@acc`
  ssubmitter=`echo $submitter|perl -pe 's/^(\w{6}).*/$1/'`

  if [ ! -f ${sra}.sra ] && [ ! -f ${sra}.fastq ] && [ ! -f ${sra}_1.fastq ] && [ ! -f ${sra}_2.fastq ]
  then
    aria2c -x 10 "ftp://ftp.ddbj.nig.ac.jp/ddbj_database/dra/sralite/ByExp/litesra/${ssexpt}/${sexpt}/${expt}/${sra}/${sra}.sra"
  fi
  if [ -f ${sra}.sra ] && [ ! -f ${sra}.fastq ] && [ ! -f ${sra}_1.fastq ] && [ ! -f ${sra}_2.fastq ]
  then
    fasterq-dump -p -e 60 --split-files ${sra}.sra
    rm ${sra}.sra
  fi
  if  [ ! -f ${sra}.fastq ] && [ ! -f ${sra}_1.fastq ] && [ ! -f ${sra}_2.fastq ]
  then
    aria2c -x 10 "https://ddbj.nig.ac.jp/public/ddbj_database/dra/fastq/${ssubmitter}/${submitter}/${expt}/${sra}.fastq.bz2"
    aria2c -x 10 "https://ddbj.nig.ac.jp/public/ddbj_database/dra/fastq/${ssubmitter}/${submitter}/${expt}/${sra}_1.fastq.bz2"
    aria2c -x 10 "https://ddbj.nig.ac.jp/public/ddbj_database/dra/fastq/${ssubmitter}/${submitter}/${expt}/${sra}_2.fastq.bz2"
    aria2c -x 10 "https://ddbj.nig.ac.jp/public/ddbj_database/dra/fastq/${ssubmitter}/${submitter}/${expt}/${sra}_3.fastq.bz2"
    pbzip2 -d -p60 ${sra}*.fastq.bz2
  fi
  if [ ! -f ${sra}.fastq ] && [ ! -f ${sra}_1.fastq ] && [ ! -f ${sra}_2.fastq ]
  then
    prefetch ${sra}
    fasterq-dump -e 60 -p --split-files ${sra}/${sra}.sra
    rm -r ${sra}
  fi
  if [ ! -f ${sra}.fastq ] && [ ! -f ${sra}_1.fastq ] && [ ! -f ${sra}_2.fastq ]
  then
    fasterq-dump -p --split-files ${sra}
  fi
  if [ -f ${sra}_1.fastq ] && [ -f ${sra}_2.fastq ] && [ ! -f ${species}_${sra}.bam ]
  then
    fastp -i ${sra}_1.fastq -o ${sra}_R1.fastq -I ${sra}_2.fastq -O ${sra}_R2.fastq -w 3
  elif [ -f ${sra}_2.fastq ] && [ -f ${sra}_3.fastq ] && [ ! -f ${species}_${sra}.bam ]
  then
    fastp -i ${sra}_2.fastq -o ${sra}_R1.fastq -I ${sra}_3.fastq -O ${sra}_R2.fastq -w 3
  elif ([ -f ${sra}_1.fastq ] || [ -f ${sra}_2.fastq ] || [ -f ${sra}.fastq ]) && [ ! -f ${species}_${sra}.bam ]
  then
    fastp -i ${sra}*.fastq -o ${sra}_R.fastq -w 3
  fi

  if [ -f ${sra}_R.fastq ] || [ -f ${sra}_R1.fastq ]; then
    echo "Mapping $sra with STAR..."
    
    # マッピング実行（1サンプルに60スレッド割り当て）
    # Ribo-seqとRNA-seq両方に対応するため、TranscriptomeSAMを出力しておく
    fastq_input="${sra}_R.fastq"
    [ -f ${sra}_R1.fastq ] && fastq_input="${sra}_R1.fastq ${sra}_R2.fastq"

    STAR --runThreadN 60 --genomeDir ${species}_STAR_index \
       --readFilesIn $fastq_input --outFileNamePrefix ${species}_${sra}_ \
       --outSAMtype BAM SortedByCoordinate --quantMode TranscriptomeSAM GeneCounts \
       --limitBAMsortRAM 30000000000  # ← ここを追加（約30GB）

    # 5. データ種別による分岐
    if [ "$data_type" == "Ribo" ]; then
      # --- Ribo-seqの場合: RiboCodeでコドン集計 ---
      # RiboCodeの -g には、prepare_transcriptsで作った「ディレクトリ」を指定する
      #metaplots -a ${species}_RiboCode_annot \
      #    -r ${species}_${sra}_Aligned.toTranscriptome.out.bam \
      #    -o ${species}_${sra}_config
    　#RiboCode -a ${species}_RiboCode_annot \
      #         -c ${species}_${sra}_config_pre_config.txt \
      #        -o ${RESULT_DIR}/${species}_${sra}_ribocode
      #Neither metaplots nor RiboCode worked in the initial trial and the loop given below was to fix it
       
    else
      # --- RNA-seqの場合: featureCountsで遺伝子集計 ---
      # STARの標準出力(GeneCounts)も使えるが、詳細設定のためfeatureCountsを実行
      featureCounts -T 60 -g ID -t gene -a ${species}*.gff3 \
                    -o ${RESULT_DIR}/${species}_${sra}_gene.count ${species}_${sra}_Aligned.sortedByCoord.out.bam
      # TPM変換 (既存のcount2tpm.plを利用)
      count2tpm.pl -g 1 -l 6 -c 7 ${RESULT_DIR}/${species}_${sra}_gene.count > ${RESULT_DIR}/${species}_${sra}_gene.tpm
    fi

    # 7. ファイル整理とBAMの外部移動
    # 巨大なBAMを外部ストレージへ移動
    mv ${species}_${sra}_Aligned.toTranscriptome.out.bam ${species}_${sra}_Aligned.sortedByCoord.out.bam ${EXTERNAL_BAM_DIR}/
    # 不要な中間ファイル（ゲノム座標BAM等）を削除してディスクを空ける

    rm -rf ${species}_${sra}__STARtmp
    rm ${sra}*.fastq
  fi
done

for i in $(awk -F"\t" 'BEGIN{OFS=","} {if($10 == 1 && $2 == "Arabidopsis thaliana" && $11 == "y"){gsub(/.*\(|\)/, "", $2); gsub(" ", "_", $2); gsub(" ", "_", $6); print $2,$3,$6}}' further_Ribo-Seq_data.txt)
do
  species=$(echo $i | cut -d',' -f1)
  sra=$(echo $i | cut -d',' -f2)
  data_type=$(echo $i | perl -pe 's/.*(RNA|Ribo).*/$1/')
  
  if [ "$data_type" != "Ribo" ]; then
    continue
  fi

  for j in pt mt; do
    echo "Processing Ribo-Seq: ${species}_${sra} (${j})"
    
    CONFIG="${species}_${sra}_${j}_config_pre_config.txt"
    ANNOT="At_${j}_RiboCode_annot"
    BAM="${EXTERNAL_BAM_DIR}/${species}_${sra}_Aligned.toTranscriptome.out.bam"
    OUT_TSV="${RESULT_DIR}/${species}_${sra}_ribodensity_${j}.tsv"
    DB="At_${j}_targetcds.db"

    # 1. フィルタリング用のIDリストを作成
    # アノテーション内の transcripts.txt からID（第1列）を抽出
    GENE_LIST="list_${j}.txt"
    if [ -f "${ANNOT}/transcripts_cds.txt" ]; then
      cut -f1 "${ANNOT}/transcripts_cds.txt" > "$GENE_LIST"
    else
      echo "Error: Annotation ${ANNOT} not found."
      continue
    fi

    # 2. BAMヘッダーとリードをアノテーションに合わせて完全に抽出
    # ヘッダー(@SQ)を抽出し、データ行もリストにマッチするものだけに絞る
    TEMP_BAM="${j}.temp.bam"
    samtools view -h "$BAM" | \
    awk -v list="$GENE_LIST" 'BEGIN{while((getline < list)>0) l[$1]=1} /^@SQ/{split($2,a,":"); if(l[a[2]]!=1) next} /^@/ || l[$3]==1' | \
    samtools view -bS - | \
    samtools sort -o "$TEMP_BAM" 
    samtools index "$TEMP_BAM"

    # 3. オフセット確認用のPDF生成（設定ファイルがない場合のみ）
    if [ ! -f "$CONFIG" ]; then
      metaplots -a "$ANNOT" \
                -r "$TEMP_BAM" \
                -o "${species}_${sra}_${j}_config" \
                -m 20 -M 40
    fi
    
    # 4. コドン情報の抽出実行
    if python3 extract_codon_stalling_ptmt.py "$CONFIG" "$ANNOT" "$TEMP_BAM" "$OUT_TSV" -d $DB --all-transcripts; then
      echo "Successfully created ${OUT_TSV}"
    else
      echo "Failed processing ${species}_${sra} (${j})"
    fi

    # 一時ファイルの削除（任意）
    rm "$TEMP_BAM" "${TEMP_BAM}.bai" "$GENE_LIST"
  done
done
